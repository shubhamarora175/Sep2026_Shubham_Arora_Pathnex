Ansible - Install and enable Nginx
. name: Install and start Nginx on Pathnex
hosts: all
became:yes

tasks:
	-name : Install nginx
  	yum:
	name:nginx
	state: present

	-name:Enable nginx
	service:
	 name:nginx
	state:started
	enabled:yes
